<script>

  /*

   * IMPORTANT

   */

  const GAS_WEB_APP_URL =
    "https://script.google.com/macros/s/AKfycbxKPOKY98yBA9Y-lZOiOa9Wzxx7a2nP6fwIE74yYKZGHjPh9QcpuB7pBcK8UQaOegON9g/exec";


  /*
   * =====================================================
   * ELEMENTS
   * =====================================================
   */

  const video =
    document.getElementById("video");

  const canvas =
    document.getElementById("canvas");

  const ctx =
    canvas.getContext(
      "2d",
      {
        willReadFrequently: true
      }
    );

  const cameraStatus =
    document.getElementById("cameraStatus");

  const messageBox =
    document.getElementById("message");


  /*
   * =====================================================
   * VARIABLES
   * =====================================================
   */

  let stream = null;

  let scanning = false;

  let processing = false;

  let currentFacingMode = "environment";

  let lastQR = "";

  let lastScanTime = 0;


  /*
   * =====================================================
   * START CAMERA
   * =====================================================
   */

  async function startCamera() {

    if (processing) {
      return;
    }

    hideMessage();

    stopCamera();

    if (
      !navigator.mediaDevices ||
      !navigator.mediaDevices.getUserMedia
    ) {

      showMessage(
        "CAMERA NOT SUPPORTED",
        "error"
      );

      return;
    }

    try {

      stream =
        await navigator.mediaDevices.getUserMedia({

          video: {

            facingMode: {
              ideal: currentFacingMode
            },

            width: {
              ideal: 1280
            },

            height: {
              ideal: 720
            }

          },

          audio: false

        });

      video.srcObject =
        stream;

      await video.play();

      scanning = true;

      cameraStatus.innerHTML =
        "🟢 CAMERA ON";

      cameraStatus.style.background =
        "#d1e7dd";

      cameraStatus.style.color =
        "#0f5132";

      requestAnimationFrame(
        scanCamera
      );

    }

    catch (error) {

      console.error(error);

      cameraStatus.innerHTML =
        "🔴 CAMERA ERROR";

      cameraStatus.style.background =
        "#f8d7da";

      cameraStatus.style.color =
        "#842029";

      showMessage(

        "CAMERA PERMISSION BLOCKED<br>" +

        "<small>" +

        "Allow camera access in your browser, " +

        "then press START CAMERA again." +

        "</small>",

        "error"

      );

    }

  }


  /*
   * =====================================================
   * STOP CAMERA
   * =====================================================
   */

  function stopCamera() {

    scanning = false;

    if (stream) {

      stream
        .getTracks()
        .forEach(function(track) {

          track.stop();

        });

      stream = null;
    }

    video.srcObject = null;

    cameraStatus.innerHTML =
      "🔴 CAMERA OFF";

    cameraStatus.style.background =
      "#e9ecef";

    cameraStatus.style.color =
      "#212529";
  }


  /*
   * =====================================================
   * SWITCH CAMERA
   * =====================================================
   */

  async function switchCamera() {

    if (processing) {
      return;
    }

    currentFacingMode =
      currentFacingMode === "environment"
        ? "user"
        : "environment";

    await startCamera();
  }


  /*
   * =====================================================
   * SCAN CAMERA
   * =====================================================
   */

  function scanCamera() {

    if (
      !scanning ||
      processing
    ) {
      return;
    }

    if (
      video.readyState !==
      video.HAVE_ENOUGH_DATA
    ) {

      requestAnimationFrame(
        scanCamera
      );

      return;
    }

    if (
      video.videoWidth <= 0 ||
      video.videoHeight <= 0
    ) {

      requestAnimationFrame(
        scanCamera
      );

      return;
    }

    canvas.width =
      video.videoWidth;

    canvas.height =
      video.videoHeight;

    ctx.drawImage(

      video,

      0,

      0,

      canvas.width,

      canvas.height

    );

    const imageData =
      ctx.getImageData(

        0,

        0,

        canvas.width,

        canvas.height

      );

    const code =
      jsQR(

        imageData.data,

        imageData.width,

        imageData.height,

        {
          inversionAttempts:
            "attemptBoth"
        }

      );

    if (code && code.data) {

      handleQR(
        code.data
      );

      return;
    }

    requestAnimationFrame(
      scanCamera
    );
  }


  /*
   * =====================================================
   * HANDLE QR
   * =====================================================
   */

  function handleQR(qrValue) {

    if (!qrValue) {
      return;
    }

    qrValue =
      String(qrValue).trim();

    if (!qrValue) {
      return;
    }

    const now =
      Date.now();

    /*
     * Prevent duplicate scan
     */

    if (

      qrValue === lastQR &&

      now - lastScanTime < 3000

    ) {

      return;
    }

    lastQR =
      qrValue;

    lastScanTime =
      now;

    processing = true;

    scanning = false;

    stopCamera();

    showMessage(
      "QR CODE FOUND<br><br>PROCESSING...",
      "info"
    );

    /*
     * =================================================
     * SEND QR TO GOOGLE APPS SCRIPT
     * =================================================
     *
     * We use normal browser navigation instead of
     * fetch/google.script.run.
     *
     * This works from an external HTTPS page.
     */

    if (

      !GAS_WEB_APP_URL ||

      GAS_WEB_APP_URL.indexOf(
        "PASTE_YOUR"
      ) !== -1

    ) {

      processing = false;

      showMessage(

        "WEB APP URL NOT SET<br>" +

        "<small>" +

        "Please put your Google Apps Script " +

        "Web App URL inside GAS_WEB_APP_URL."

        + "</small>",

        "error"

      );

      return;
    }

    const targetURL =
      GAS_WEB_APP_URL +

      "?qr=" +

      encodeURIComponent(
        qrValue
      );

    window.location.href =
      targetURL;
  }


  /*
   * =====================================================
   * MESSAGE
   * =====================================================
   */

  function showMessage(
    message,
    type
  ) {

    messageBox.innerHTML =
      message;

    messageBox.className =
      "";

    messageBox.classList.add(
      type
    );

    messageBox.style.display =
      "block";
  }


  function hideMessage() {

    messageBox.innerHTML =
      "";

    messageBox.className =
      "";

    messageBox.style.display =
      "none";
  }


  /*
   * =====================================================
   * CLEANUP
   * =====================================================
   */

  window.addEventListener(
    "beforeunload",
    function() {
      stopCamera();
    }
  );
